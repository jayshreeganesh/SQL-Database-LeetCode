# First Letter Capitalization II: Solution Implementations

## 1. PostgreSQL

Utilizes `UNNEST` with `WITH ORDINALITY` to easily tokenize the string while maintaining the original order.

```sql
SELECT 
    content_id, 
    content_text AS original_text,
    (
        SELECT STRING_AGG(
            CASE 
                -- Strict pattern match for letters separated by hyphens
                WHEN word ~ '^[a-zA-Z]+(-[a-zA-Z]+)+$' THEN 
                    (SELECT STRING_AGG(INITCAP(part), '-' ORDER BY ord)
                     FROM UNNEST(STRING_TO_ARRAY(word, '-')) WITH ORDINALITY AS p(part, ord))
                
                -- Fallback: Capitalize first character, lowercase the rest
                ELSE 
                    UPPER(LEFT(word, 1)) || LOWER(SUBSTRING(word FROM 2))
            END, 
            ' ' ORDER BY ord
        )
        FROM UNNEST(STRING_TO_ARRAY(user_content.content_text, ' ')) WITH ORDINALITY AS t(word, ord)
    ) AS converted_text
FROM user_content;

```

## 2. MySQL

Uses a `RECURSIVE CTE` to generate word positions and handle tokenization through `SUBSTRING_INDEX`.

```sql
WITH RECURSIVE numbers AS (
    SELECT 1 AS n UNION ALL SELECT n + 1 FROM numbers WHERE n < 100
),
words AS (
    SELECT content_id, content_text,
           SUBSTRING_INDEX(SUBSTRING_INDEX(content_text, ' ', n), ' ', -1) AS word,
           n AS position
    FROM user_content
    JOIN numbers ON n <= LENGTH(content_text) - LENGTH(REPLACE(content_text, ' ', '')) + 1
),
transformed AS (
    SELECT content_id, content_text, position,
        CASE
            WHEN word REGEXP '^[A-Za-z]+(-[A-Za-z]+)+$' THEN
                (SELECT GROUP_CONCAT(CONCAT(UPPER(SUBSTRING(part, 1, 1)), LOWER(SUBSTRING(part, 2))) 
                 ORDER BY n SEPARATOR '-')
                 FROM (SELECT SUBSTRING_INDEX(SUBSTRING_INDEX(word, '-', n), '-', -1) AS part, n
                       FROM numbers WHERE n <= LENGTH(word) - LENGTH(REPLACE(word, '-', '')) + 1) parts)
            WHEN word REGEXP '^[A-Za-z]' THEN CONCAT(UPPER(SUBSTRING(word, 1, 1)), LOWER(SUBSTRING(word, 2)))
            ELSE word
        END AS converted_word
    FROM words
)
SELECT content_id, content_text AS original_text,
       GROUP_CONCAT(converted_word ORDER BY position SEPARATOR ' ') AS converted_text
FROM transformed
GROUP BY content_id, content_text;

```

## 3. MS SQL Server

Leverages the `OPENJSON` function as a robust method to split strings into ordered tokens.

```sql
WITH SpaceSplit AS (
    SELECT content_id, content_text, CAST([key] AS INT) AS word_pos, [value] AS word
    FROM user_content
    CROSS APPLY OPENJSON('["' + REPLACE(REPLACE(content_text, '\', '\\'), ' ', '","') + '"]')
),
HyphenSplit AS (
    SELECT s.content_id, s.word_pos, h.[value] AS part, h.[key] AS part_pos
    FROM SpaceSplit s
    CROSS APPLY OPENJSON(CASE WHEN s.word LIKE '%-%' AND s.word NOT LIKE '%[^a-zA-Z-]%' 
                              THEN '["' + REPLACE(REPLACE(s.word, '\', '\\'), '-', '","') + '"]'
                              ELSE '["' + REPLACE(s.word, '\', '\\') + '"]' END) h
),
Capitalized AS (
    SELECT content_id, word_pos, part_pos,
           UPPER(SUBSTRING(part, 1, 1)) + LOWER(SUBSTRING(part, 2, LEN(part))) AS cap_part
    FROM HyphenSplit
),
Recombined AS (
    SELECT content_id, word_pos, STRING_AGG(cap_part, '-') WITHIN GROUP (ORDER BY part_pos) AS final_word
    FROM Capitalized GROUP BY content_id, word_pos
)
SELECT content_id, content_text AS original_text, STRING_AGG(final_word, ' ') WITHIN GROUP (ORDER BY word_pos) AS converted_text
FROM Recombined GROUP BY content_id, content_text;

```

## 4. Oracle

Uses a `Recursive CTE` for tokenization and `LISTAGG` for re-aggregation.

```sql
WITH word_bounds (content_id, content_text, word_pos, start_pos, space_pos) AS (
    SELECT content_id, content_text, 1, 1, INSTR(content_text, ' ', 1) FROM user_content
    UNION ALL
    SELECT content_id, content_text, word_pos + 1, space_pos + 1, INSTR(content_text, ' ', space_pos + 1)
    FROM word_bounds WHERE space_pos > 0
),
transformed_words AS (
    SELECT content_id, content_text, word_pos,
           CASE WHEN REGEXP_LIKE(word, '^[A-Za-z]+(-[A-Za-z]+)+$') THEN INITCAP(word)
                WHEN LENGTH(word) > 0 THEN UPPER(SUBSTR(word, 1, 1)) || LOWER(SUBSTR(word, 2))
                ELSE '' END AS converted_word
    FROM (SELECT content_id, content_text, word_pos,
                 CASE WHEN space_pos = 0 THEN SUBSTR(content_text, start_pos)
                      ELSE SUBSTR(content_text, start_pos, space_pos - start_pos) END AS word
          FROM word_bounds)
)
SELECT content_id, content_text AS original_text,
       LISTAGG(converted_word, ' ') WITHIN GROUP (ORDER BY word_pos) AS converted_text
FROM transformed_words GROUP BY content_id, content_text;

```

## 5. Pandas

Uses Python's regex engine and the `apply()` method for efficient string manipulation.

```python
import pandas as pd
import re

def capitalize_content(user_content: pd.DataFrame) -> pd.DataFrame:
    def transform_word(word):
        # Strict regex for letters separated by single hyphens
        if re.match(r'^[A-Za-z]+(-[A-Za-z]+)+$', word):
            return '-'.join(part.capitalize() for part in word.split('-'))
        return word.capitalize()

    def transform_text(text):
        return ' '.join(transform_word(word) for word in text.split(' '))
        
    user_content['original_text'] = user_content['content_text']
    user_content['converted_text'] = user_content['content_text'].apply(transform_text)
    return user_content[['content_id', 'original_text', 'converted_text']]

```
