## 1. MySQL

This approach leverages a **Recursive CTE** paired with `FIND_IN_SET()` to identify precise structural matches rather than utilizing standard substring searches.

```sql
WITH RECURSIVE Hierarchy AS (
    -- Base Case: Start with the top-level CEO
    SELECT 
        employee_id,
        employee_name,
        manager_id,
        salary,
        1 AS level,
        CAST(employee_id AS CHAR(1000)) AS path
    FROM Employees
    WHERE manager_id IS NULL

    UNION ALL

    -- Recursive Case: Chain downline employees to their direct managers
    SELECT 
        e.employee_id,
        e.employee_name,
        e.manager_id,
        e.salary,
        h.level + 1 AS level,
        CONCAT(h.path, ',', e.employee_id) AS path
    FROM Employees e
    INNER JOIN Hierarchy h ON e.manager_id = h.employee_id
),
RollupStats AS (
    -- Map subordinates accurately by evaluating CSV lists via FIND_IN_SET
    SELECT 
        h1.employee_id,
        COUNT(h2.employee_id) - 1 AS team_size,
        SUM(h2.salary) AS budget
    FROM Hierarchy h1
    LEFT JOIN Hierarchy h2 ON FIND_IN_SET(h1.employee_id, h2.path) > 0
    GROUP BY h1.employee_id
)
SELECT 
    h.employee_id,
    h.employee_name,
    h.level,
    r.team_size,
    r.budget
FROM Hierarchy h
INNER JOIN RollupStats r ON h.employee_id = r.employee_id
ORDER BY 
    h.level ASC, 
    r.budget DESC, 
    h.employee_name ASC;

```

## 2. MS SQL Server (T-SQL)

To avoid partial numeric ID overrides (such as ID `5` matching inside `15` or `25`), this query structures the tracking strings symmetrically with leading and trailing commas (`%,ID,%`).

```sql
WITH Hierarchy AS (
    -- Base Case: Target the CEO
    SELECT 
        employee_id,
        employee_name,
        manager_id,
        salary,
        1 AS level,
        CAST(',' + CAST(employee_id AS VARCHAR(50)) + ',' AS VARCHAR(MAX)) AS path
    FROM Employees
    WHERE manager_id IS NULL

    UNION ALL

    -- Recursive Case: Drill down the management chain
    SELECT 
        e.employee_id,
        e.employee_name,
        e.manager_id,
        e.salary,
        h.level + 1 AS level,
        CAST(h.path + CAST(e.employee_id AS VARCHAR(50)) + ',' AS VARCHAR(MAX)) AS path
    FROM Employees e
    INNER JOIN Hierarchy h ON e.manager_id = h.employee_id
),
RollupStats AS (
    -- Safely match exact, comma-bounded integer IDs
    SELECT 
        h1.employee_id,
        COUNT(h2.employee_id) - 1 AS team_size,
        SUM(h2.salary) AS budget
    FROM Hierarchy h1
    LEFT JOIN Hierarchy h2 ON h2.path LIKE '%,' + CAST(h1.employee_id AS VARCHAR(50)) + ',%'
    GROUP BY h1.employee_id
)
SELECT 
    h.employee_id,
    h.employee_name,
    h.level,
    r.team_size,
    r.budget
FROM Hierarchy h
INNER JOIN RollupStats r ON h.employee_id = r.employee_id
ORDER BY 
    h.level ASC, 
    r.budget DESC, 
    h.employee_name ASC;
```

## 3. PostgreSQL

PostgreSQL avoids string parsing entirely by using native integer array types (`INT[]`) alongside the `ANY()` operator. This is the cleanest and most performant relational approach.

```sql
WITH RECURSIVE Hierarchy AS (
    -- Base Case: Instantiate an integer array with the CEO
    SELECT 
        employee_id,
        employee_name,
        manager_id,
        salary,
        1 AS level,
        ARRAY[employee_id] AS path
    FROM Employees
    WHERE manager_id IS NULL

    UNION ALL

    -- Recursive Case: Append downstream employee_ids into the array tracking chain
    SELECT 
        e.employee_id,
        e.employee_name,
        e.manager_id,
        e.salary,
        h.level + 1 AS level,
        h.path || e.employee_id AS path
    FROM Employees e
    INNER JOIN Hierarchy h ON e.manager_id = h.employee_id
),
RollupStats AS (
    -- Inspect ancestral chains natively via array elements
    SELECT 
        h1.employee_id,
        COUNT(h2.employee_id) - 1 AS team_size,
        SUM(h2.salary) AS budget
    FROM Hierarchy h1
    LEFT JOIN Hierarchy h2 ON h1.employee_id = ANY(h2.path)
    GROUP BY h1.employee_id
)
SELECT 
    h.employee_id,
    h.employee_name,
    h.level,
    r.team_size,
    r.budget
FROM Hierarchy h
INNER JOIN RollupStats r ON h.employee_id = r.employee_id
ORDER BY 
    h.level ASC, 
    r.budget DESC, 
    h.employee_name ASC;

```

## 4. Oracle PL/SQL

This solution handles Oracle-specific structural behaviors: it safely bypasses the reserved keyword `LEVEL` by computing under `lvl` and forces a clean `VARCHAR2(4000)` casting buffer.

```sql
WITH Hierarchy (employee_id, employee_name, manager_id, salary, lvl, path) AS (
    -- Base Case
    SELECT 
        employee_id,
        employee_name,
        manager_id,
        salary,
        1 AS lvl,
        CAST(',' || employee_id || ',' AS VARCHAR2(4000)) AS path
    FROM Employees
    WHERE manager_id IS NULL

    UNION ALL

    -- Recursive Case
    SELECT 
        e.employee_id,
        e.employee_name,
        e.manager_id,
        e.salary,
        h.lvl + 1 AS lvl,
        CAST(h.path || e.employee_id || ',' AS VARCHAR2(4000)) AS path
    FROM Employees e
    INNER JOIN Hierarchy h ON e.manager_id = h.employee_id
),
RollupStats AS (
    SELECT 
        h1.employee_id,
        COUNT(h2.employee_id) - 1 AS team_size,
        SUM(h2.salary) AS budget
    FROM Hierarchy h1
    LEFT JOIN Hierarchy h2 ON h2.path LIKE '%,' || h1.employee_id || ',%'
    GROUP BY h1.employee_id
)
SELECT 
    h.employee_id,
    h.employee_name,
    h.lvl AS "LEVEL",
    r.team_size,
    r.budget
FROM Hierarchy h
INNER JOIN RollupStats r ON h.employee_id = r.employee_id
ORDER BY 
    "LEVEL" ASC, 
    r.budget DESC, 
    h.employee_name ASC;

```

## 5. Python (Pandas)

A highly readable approach that builds an explicit upward reporting map utilizing fast $O(1)$ dictionary lookups to find org depth and track rolling financials.

```python
import pandas as pd

def analyze_organization_hierarchy(employees: pd.DataFrame) -> pd.DataFrame:
    path_map = {}
    level_map = {}
    
    # Map out direct links for fast ancestor lookups
    manager_dict = dict(zip(employees['employee_id'], employees['manager_id']))
    
    # Establish total ancestor paths and level indicators
    for emp_id in employees['employee_id']:
        path = []
        curr = emp_id
        while pd.notna(curr):
            path.append(int(curr))
            curr = manager_dict.get(curr)
        path_map[emp_id] = path
        level_map[emp_id] = len(path)
        
    employees['level'] = employees['employee_id'].map(level_map)
    
    # Calculate rollup dimensions
    team_sizes = []
    budgets = []
    
    for emp_id in employees['employee_id']:
        subordinates = [
            other_id for other_id, path in path_map.items() 
            if emp_id in path
        ]
        team_sizes.append(len(subordinates) - 1)
        
        sub_salaries = employees[employees['employee_id'].isin(subordinates)]['salary'].sum()
        budgets.append(sub_salaries)
        
    employees['team_size'] = team_sizes
    employees['budget'] = budgets
    
    # Filter and Sort output based on LeetCode specifications
    result = employees[['employee_id', 'employee_name', 'level', 'team_size', 'budget']]
    return result.sort_values(
        by=['level', 'budget', 'employee_name'], 
        ascending=[True, False, True]
    )

```
