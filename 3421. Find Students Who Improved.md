## 1. MySQL
Uses Window Functions (`ROW_NUMBER`) within a Common Table Expression (CTE) to identify the first and last exams.

```sql
WITH RankedScores AS (
    SELECT 
        student_id,
        subject,
        score,
        exam_date,
        COUNT(*) OVER(PARTITION BY student_id, subject) as exam_count,
        ROW_NUMBER() OVER(PARTITION BY student_id, subject ORDER BY exam_date ASC) as rn_first,
        ROW_NUMBER() OVER(PARTITION BY student_id, subject ORDER BY exam_date DESC) as rn_latest
    FROM Scores
),
FilteredScores AS (
    SELECT 
        student_id,
        subject,
        MAX(CASE WHEN rn_first = 1 THEN score END) AS first_score,
        MAX(CASE WHEN rn_latest = 1 THEN score END) AS latest_score
    FROM RankedScores
    WHERE exam_count >= 2
    GROUP BY student_id, subject
)
SELECT student_id, subject, first_score, latest_score
FROM FilteredScores
WHERE latest_score > first_score
ORDER BY student_id ASC, subject ASC;

```

## 5. Pandas

An idiomatic, vectorized Python approach using `groupby`, `transform`, and aggregation.

```python
import pandas as pd

def find_students_who_improved(scores: pd.DataFrame) -> pd.DataFrame:
    # Ensure dates are datetime objects for proper chronological sorting
    scores['exam_date'] = pd.to_datetime(scores['exam_date'])
    scores = scores.sort_values(by=['student_id', 'subject', 'exam_date'])
    
    # Filter for students with at least 2 exams
    counts = scores.groupby(['student_id', 'subject'])['score'].transform('count')
    scores = scores[counts >= 2]
    
    # Group and aggregate first and last scores
    result = scores.groupby(['student_id', 'subject']).agg(
        first_score=('score', 'first'),
        latest_score=('score', 'last')
    ).reset_index()
    
    # Filter for improvement
    result = result[result['latest_score'] > result['first_score']]
    
    return result[['student_id', 'subject', 'first_score', 'latest_score']]

```
